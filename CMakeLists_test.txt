cmake_minimum_required(VERSION 3.20)

project(GismoIntegrationTest)

# Find the GISMO package
find_package(gismo REQUIRED PATHS ${CMAKE_CURRENT_SOURCE_DIR}/build NO_DEFAULT_PATH)

# Create the test executable
add_executable(integration_test test_integration.cpp)

# Link with GISMO modules
target_link_libraries(integration_test
    PRIVATE
        gismo::Core
        gismo::Math
        gismo::Geometry
)

# Set C++ standard
target_compile_features(integration_test PRIVATE cxx_std_17)

# Add include directories
target_include_directories(integration_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
